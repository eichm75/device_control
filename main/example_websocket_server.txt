#define INDEX_HTML_PATH "/spiffs/index.html"

static void initi_web_page_buffer(void)
{
    esp_vfs_spiffs_conf_t conf = {
        .base_path = "/spiffs",
        .partition_label = NULL,
        .max_files = 5,
        .format_if_mount_failed = true};

    ESP_ERROR_CHECK(esp_vfs_spiffs_register(&conf));

    memset((void *)index_html, 0, sizeof(index_html));
    struct stat st;
    if (stat(INDEX_HTML_PATH, &st))
    {
        ESP_LOGE(TAG, "index.html not found");
        return;
    }

    FILE *fp = fopen(INDEX_HTML_PATH, "r");
    if (fread(index_html, st.st_size, 1, fp) == 0)
    {
        ESP_LOGE(TAG, "fread failed");
    }
    fclose(fp);
}

esp_err_t get_req_handler(httpd_req_t *req)
{
    int response;
    if(led_state)
    {
        sprintf(response_data, index_html, "ON");
    }
    else
    {
        sprintf(response_data, index_html, "OFF");
    }
    response = httpd_resp_send(req, response_data, HTTPD_RESP_USE_STRLEN);
    return response;
}

static void ws_async_send(void *arg)
{
    httpd_ws_frame_t ws_pkt;
    struct async_resp_arg *resp_arg = arg;
    httpd_handle_t hd = resp_arg->hd;
    int fd = resp_arg->fd;

    led_state = !led_state;
    gpio_set_level(LED_PIN, led_state);
    
    char buff[4];
    memset(buff, 0, sizeof(buff));
    sprintf(buff, "%d",led_state);
    
    memset(&ws_pkt, 0, sizeof(httpd_ws_frame_t));
    ws_pkt.payload = (uint8_t *)buff;
    ws_pkt.len = strlen(buff);
    ws_pkt.type = HTTPD_WS_TYPE_TEXT;
    
    static size_t max_clients = CONFIG_LWIP_MAX_LISTENING_TCP;
    size_t fds = max_clients;
    int client_fds[max_clients];

    esp_err_t ret = httpd_get_client_list(server, &fds, client_fds);

    if (ret != ESP_OK) {
        return;
    }

    for (int i = 0; i < fds; i++) {
        int client_info = httpd_ws_get_fd_info(server, client_fds[i]);
        if (client_info == HTTPD_WS_CLIENT_WEBSOCKET) {
            httpd_ws_send_frame_async(hd, client_fds[i], &ws_pkt);
        }
    }
    free(resp_arg);
}

static esp_err_t trigger_async_send(httpd_handle_t handle, httpd_req_t *req)
{
    struct async_resp_arg *resp_arg = malloc(sizeof(struct async_resp_arg));
    resp_arg->hd = req->handle;
    resp_arg->fd = httpd_req_to_sockfd(req);
    return httpd_queue_work(handle, ws_async_send, resp_arg);
}

static esp_err_t handle_ws_req(httpd_req_t *req)
{
    if (req->method == HTTP_GET)
    {
        ESP_LOGI(TAG, "Handshake done, the new connection was opened");
        return ESP_OK;
    }

    httpd_ws_frame_t ws_pkt;
    uint8_t *buf = NULL;
    memset(&ws_pkt, 0, sizeof(httpd_ws_frame_t));
    ws_pkt.type = HTTPD_WS_TYPE_TEXT;
    esp_err_t ret = httpd_ws_recv_frame(req, &ws_pkt, 0);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "httpd_ws_recv_frame failed to get frame len with %d", ret);
        return ret;
    }

    if (ws_pkt.len)
    {
        buf = calloc(1, ws_pkt.len + 1);
        if (buf == NULL)
        {
            ESP_LOGE(TAG, "Failed to calloc memory for buf");
            return ESP_ERR_NO_MEM;
        }
        ws_pkt.payload = buf;
        ret = httpd_ws_recv_frame(req, &ws_pkt, ws_pkt.len);
        if (ret != ESP_OK)
        {
            ESP_LOGE(TAG, "httpd_ws_recv_frame failed with %d", ret);
            free(buf);
            return ret;
        }
        ESP_LOGI(TAG, "Got packet with message: %s", ws_pkt.payload);
    }

    ESP_LOGI(TAG, "frame len is %d", ws_pkt.len);

    if (ws_pkt.type == HTTPD_WS_TYPE_TEXT &&
        strcmp((char *)ws_pkt.payload, "toggle") == 0)
    {
        free(buf);
        return trigger_async_send(req->handle, req);
    }
    return ESP_OK;
}

httpd_handle_t setup_websocket_server(void)
{
    httpd_config_t config = HTTPD_DEFAULT_CONFIG();

    httpd_uri_t uri_get = {
        .uri = "/",
        .method = HTTP_GET,
        .handler = get_req_handler,
        .user_ctx = NULL};

    httpd_uri_t ws = {
        .uri = "/ws",
        .method = HTTP_GET,
        .handler = handle_ws_req,
        .user_ctx = NULL,
        .is_websocket = true};

    if (httpd_start(&server, &config) == ESP_OK)
    {
        httpd_register_uri_handler(server, &uri_get);
        httpd_register_uri_handler(server, &ws);
    }

    return server;
}

Функция setup_websocket_server() настраивает и запускает HTTP-сервер с конечной точкой WebSocket.

Функция начинается с определения переменной типа httpd_config_t с именем config и конфигурацией по умолчанию. Затем она определяет две переменные типа httpd_uri_t: uri_get и ws. uri_get — это конфигурация для 
обработки GET-запросов к корневому URL («/»), а функция обработчика для этого URI устанавливается на get_req_handler().

        .uri = "/",
        .method = HTTP_GET,
        .handler = get_req_handler,
        .user_ctx = NULL};
ws — это конфигурация для обработки запросов WebSocket. URI для этой конфигурации устанавливается на «/ws», метод для этого URI — на HTTP_GET, а функция обработчика для этого URI — на handle_ws_req(). 
Также устанавливается значение is_websocket равным true, что сообщает серверу httpd, что этот URI будет обрабатывать трафик WebSocket.

Обработчик начального HTTP-рукопожатия WebSocket
Как обсуждалось ранее, веб-серверы WebSocket начинают первоначальное взаимодействие с HTTP-рукопожатия, а затем следуют протоколу связи WebSocket. Поэтому нам необходимо настроить обработчик HTTP GET-запросов.

Функция «get_req_handler()» используется для отправки веб-страницы клиенту в ответ на HTTP-запрос. В качестве аргумента она принимает указатель на структуру «httpd_req_t», которая представляет собой HTTP-запрос.

Функция начинается с объявления переменной «response» типа «int». Затем она проверяет значение глобальной переменной «led_state». Е
сли оно истинно, функция использует функцию «sprintf()» для форматирования содержимого буфера «index_html» и строки «ON» в буфер «response_data». 
Если «led_state» ложно, функция форматирует содержимое буфера «index_html» и строку «OFF» в буфер «response_data».

Затем функция вызывает функцию «httpd_resp_send()», передавая ей в качестве аргументов указатель на аргумент «req», 
указатель на буфер «response_data» и значение «HTTPD_RESP_USE_STRLEN». Это отправляет содержимое буфера «response_data» в качестве ответа на HTTP-запрос. 
Затем функция возвращает значение, возвращаемое функцией «httpd_resp_send()».

Функция «get_req_handler()» используется в качестве обработчика запроса для корневого URI «/». В качестве аргумента она принимает указатель на структуру «httpd_req_t», которая представляет собой HTTP-запрос. 
Функция просто вызывает функцию «send_web_page()» и возвращает её значение.

esp_err_t get_req_handler(httpd_req_t *req)
{
    int response;
    if(led_state)
    {
        sprintf(response_data, index_html, "ON");
    }
    else
    {
        sprintf(response_data, index_html, "OFF");
    }
    response = httpd_resp_send(req, response_data, HTTPD_RESP_USE_STRLEN);
    return response;
}
Считывание содержимого файла index.html из SPIFSS в буфер (initi_web_page_buffer)
В коде веб-сервера ESP32 WebSocket функция initi_web_page_buffer() инициализирует буфер с именем «index_html» содержимым файла «index.html», хранящегося в файловой системе SPIFFS (SPI Flash File System).

Функция начинается с определения структуры с именем «conf» типа «esp_vfs_spiffs_conf_t» и инициализации её несколькими полями. Поле «base_path» задаёт корневой каталог файловой системы SPIFFS, поле «partition_label» задаёт метку раздела SPIFFS (в данном случае она устанавливается в NULL), поле «max_files» задаёт максимальное количество файлов, которые могут быть открыты одновременно, а поле «format_if_mount_failed» указывает, следует ли форматировать файловую систему в случае неудачного монтирования.

Затем функция вызывает функцию «esp_vfs_spiffs_register()», передавая ей в качестве аргумента указатель на структуру «conf». Эта функция монтирует файловую систему SPIFFS, используя указанную конфигурацию.

Затем функция обнуляет все байты в буфере «index_html» с помощью функции «memset()». После этого она объявляет структуру с именем «st» типа «struct stat» и проверяет, существует ли файл «index.html», используя функцию «stat()». Если файл не существует, она выводит сообщение об ошибке в лог и завершает работу функции.

static void initi_web_page_buffer(void)
{
    esp_vfs_spiffs_conf_t conf = {
        .base_path = "/spiffs",
        .partition_label = NULL,
        .max_files = 5,
        .format_if_mount_failed = true};

    ESP_ERROR_CHECK(esp_vfs_spiffs_register(&conf));

    memset((void *)index_html, 0, sizeof(index_html));
    struct stat st;
    if (stat(INDEX_HTML_PATH, &st))
    {
        ESP_LOGE(TAG, "index.html not found");
        return;
    }

    FILE *fp = fopen(INDEX_HTML_PATH, "r");
    if (fread(index_html, st.st_size, 1, fp) == 0)
    {
        ESP_LOGE(TAG, "fread failed");
    }
    fclose(fp);
}
